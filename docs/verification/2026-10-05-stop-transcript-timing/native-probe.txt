import json,sys,time
from pathlib import Path
p=json.load(sys.stdin); t=Path(p["transcript_path"]); out=[]
for delay in (0,0.05,0.15,0.5,1.0):
    time.sleep(delay-(out[-1]["delay"] if out else 0))
    rows=[json.loads(l) for l in t.read_text().splitlines() if l.strip()] if t.exists() else []
    texts=[]
    for r in rows:
        if r.get("type")=="assistant":
            c=r["message"]["content"]; texts.append(c if isinstance(c,str) else " ".join(x.get("text","") for x in c if isinstance(x,dict)))
    out.append({"delay":delay,"assistant_texts":texts,"candidate_present":p["last_assistant_message"] in texts})
Path(sys.argv[1]).write_text(json.dumps(out))
